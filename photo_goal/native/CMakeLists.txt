cmake_minimum_required(VERSION 3.20)
project(photo_slam_live LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
find_package(Python COMPONENTS Interpreter Development REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
find_package(Torch REQUIRED)
find_package(OpenCV 4 REQUIRED)
find_package(Eigen3 REQUIRED)
if(NOT PHOTO_SLAM_SOURCE)
  message(FATAL_ERROR "Set PHOTO_SLAM_SOURCE to the prepared pinned checkout")
endif()
execute_process(COMMAND git -C ${PHOTO_SLAM_SOURCE} rev-parse HEAD OUTPUT_VARIABLE PIN OUTPUT_STRIP_TRAILING_WHITESPACE)
if(NOT PIN STREQUAL "f8bfb2f0809c003ccc3fd577dc43c576fcafa4ac")
  message(FATAL_ERROR "Photo-SLAM revision mismatch")
endif()
file(READ "${PHOTO_SLAM_SOURCE}/include/gaussian_mapper.h" MAPPER_HEADER)
if(NOT MAPPER_HEADER MATCHES "bool liveStep")
  message(FATAL_ERROR "Run prepare_source.py before building Photo-SLAM and this bridge")
endif()
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${TORCH_CXX_FLAGS}")
pybind11_add_module(photo_slam_live photo_live.cpp)
target_include_directories(photo_slam_live PRIVATE ${PHOTO_SLAM_SOURCE} ${PHOTO_SLAM_SOURCE}/ORB-SLAM3/include ${PHOTO_SLAM_SOURCE}/ORB-SLAM3/include/CameraModels ${PHOTO_SLAM_SOURCE}/ORB-SLAM3/Thirdparty/Sophus ${OpenCV_INCLUDE_DIRS})
target_link_libraries(photo_slam_live PRIVATE ${PHOTO_SLAM_SOURCE}/lib/libgaussian_mapper.so ${PHOTO_SLAM_SOURCE}/ORB-SLAM3/lib/libORB_SLAM3.so ${TORCH_LIBRARIES} ${OpenCV_LIBS} Eigen3::Eigen)
